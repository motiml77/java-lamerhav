### הורשה תקינה — בלי אזהרות שווא
public class Zoo {
  public static void main(String[] args) {
    Animal[] zoo = { new Dog("Rexy", "Boxer"), new Cat("Mitzi") };
    for (int i = 0; i < zoo.length; i++) {
      System.out.println(zoo[i]);
      if (zoo[i] instanceof Dog) System.out.println(((Dog) zoo[i]).bark());
    }
    for (Animal a : zoo) {
      if (a instanceof Dog) System.out.println("dog");
      else if (a instanceof Animal) System.out.println("animal");
    }
    Animal pet = zoo[0];
    Dog d = (Dog) pet;
    System.out.println(d.bark() + pet.say());
  }
}
abstract class Animal {
  private String name;
  public Animal(String name) { this.name = name; }
  public String getName() { return name; }
  public abstract String say();
  public String toString() { return "Animal " + name; }
}
class Dog extends Animal {
  private String breed;
  public Dog(String name, String breed) { super(name); this.breed = breed; }
  public String say() { return "Woof"; }
  public String bark() { return say() + "!"; }
  public String toString() { return super.toString() + " " + breed; }
}
class Cat extends Animal {
  public Cat(String name) { super(name); }
  public String say() { return "Meow"; }
}
